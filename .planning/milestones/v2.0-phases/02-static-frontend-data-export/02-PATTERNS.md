```python
def parse_sssom_metadata(path: Path) -> dict:
    """Extract YAML metadata header from SSSOM file."""
    header_lines = []
    for line in path.read_text(encoding="utf-8").splitlines():
        if not line.startswith("#"):
            break
        header_lines.append(line[1:])

    if not header_lines:
        return {}

    loaded = yaml.safe_load("\n".join(header_lines))
    return loaded if isinstance(loaded, dict) else {}
```

```javascript
export async function loadGraphData(db, jsonUrl) {
  const res = await fetch(jsonUrl);
  if (!res.ok) {
    throw new Error(`Failed to fetch graph data (${res.status}): ${jsonUrl}`);
  }
  const data = await res.json();

  for (const standard of data.standards) {
    await db.create(`standard:${standard.id}`, { name: standard.name });
  }
  for (const crosswalk of data.crosswalks) {
    await db.create(`crosswalk:${crosswalk.id}`, {
      source: `standard:${crosswalk.source}`,
      target: `standard:${crosswalk.target}`,
      rules: crosswalk.rules,
    });
  }
  return data;
}
```

## Shared Patterns

### Build-time stdlib-only script conventions
**Source:** `scripts/build_pages.py` lines 1-21, 388-390, 897-899
**Apply to:** `scripts/export_crosswalk_json.py`
- Docstring declares stdlib-only intent.
- `ROOT = Path(__file__).resolve().parents[1]` for repo-root-relative paths.
- Output write: `path.parent.mkdir(parents=True, exist_ok=True)` + `path.write_text(..., encoding="utf-8")`.
- `if __name__ == "__main__":` calls a single `build()`/`export()` function and prints the output path.

### SSSOM parsing: headers + rows + comment JSON
**Sources:** `scripts/export_graph.py` lines 17-29 (YAML header via `yaml.safe_load`), `src/m2s3om_graph/sssom.py` lines 136-143 + 225-271 (DictReader + comment JSON with safe defaults), `scripts/build_pages.py` lines 61-73 (`read_sssom_file` header-index scan)
**Apply to:** `scripts/export_crosswalk_json.py`, `tests/test_export_crosswalk_json.py`
- Never trust `comment` — wrap `json.loads` in try/except, default `{}`.
- `confidence` default `0.8`, clamped `[0.0, 1.0]`.
- `semantic_loss` default: `True` when `mapping_type == missing`; `ambiguity` default: `True` for `conditional`/`aggregation` (sssom.py lines 200-215).
- Use `yaml.safe_load` (never unsafe) for headers.

### Strategy inference (deterministic / llm / fallback)
**Source:** RESEARCH.md lines 445-451 (verified against `exports/pipeline/latest/rdamsc_stats_summary.json` strategy_counts)
**Apply to:** `scripts/export_crosswalk_json.py` + `tests/test_export_crosswalk_json.py`
```python
def _infer_strategy(comment: dict) -> str:
    mt = comment.get("mapping_type", "direct")
    if mt in ("conditional", "aggregation"):
        return "llm"
    if mt == "missing":
        return "fallback"
    return "deterministic"
```
